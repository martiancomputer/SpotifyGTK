option('profile',
  type: 'combo',
  choices: ['stable', 'nightly'],
  value: 'stable',
  description: 'Build track: stable (Pulse/ALSA) or nightly (+ native PipeWire)')

option('enable_pipewire',
  type: 'boolean',
  value: false,
  description: 'Compile native PipeWire audio output (nightly default: true). Runtime-probed regardless -- falls through to Pulse/ALSA if not actually present.')

option('tests',
  type: 'boolean',
  value: true,
  description: 'Build unit tests')

option('enable_gui',
  type: 'boolean',
  value: true,
  description: 'Build the GTK4 native-player shell. Disable for headless engine/harness builds.')

option('allow_old_gtk',
  type: 'boolean',
  value: false,
  description: 'Build against GTK older than the UI was designed for. Off by default so an unnoticed downgrade cannot ship as a broken-looking build.')

option('allow_old_libadwaita',
  type: 'boolean',
  value: false,
  description: 'Build against libadwaita older than the UI was designed for. GTK has its own independent version check.')

option('verbose_logging',
  type: 'boolean',
  value: false,
  description: 'Compile in per-operation diagnostics (catalog stage timing, ' +
               'cover/cache ownership, page navigation, five-second memory/page sampling, collection writes). Off by default because ' +
               'they are noisy; -Dverbose_logging=true keeps them in the ' +
               'binary rather than deleting them from the source every time ' +
               'a bug is closed.')
